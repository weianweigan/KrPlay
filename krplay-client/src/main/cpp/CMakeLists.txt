cmake_minimum_required(VERSION 3.22.1)
project(krplay_decoder)
set(CMAKE_CXX_STANDARD 17)
set(FFMPEG "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.local/ffmpeg/android-arm64")
foreach(lib avcodec avutil swscale)
  add_library(${lib} SHARED IMPORTED)
  set_target_properties(${lib} PROPERTIES IMPORTED_LOCATION "${FFMPEG}/lib/lib${lib}.so")
endforeach()
add_library(krplay_decoder SHARED software_decoder.cpp)
target_include_directories(krplay_decoder PRIVATE "${FFMPEG}/include")
target_link_libraries(krplay_decoder avcodec avutil swscale android log)
target_link_options(krplay_decoder PRIVATE "-Wl,-z,max-page-size=16384")
